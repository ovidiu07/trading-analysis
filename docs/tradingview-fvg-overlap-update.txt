f_getOverlapIndex(float newTop, float newBottom, array<float> drawnTops, array<float> drawnBottoms) =>
    int overlapIndex = 0
    if array.size(drawnTops) > 0
        for i = 0 to array.size(drawnTops) - 1
            float existingTop = array.get(drawnTops, i)
            float existingBottom = array.get(drawnBottoms, i)
            bool overlaps = newBottom <= existingTop and newTop >= existingBottom
            if overlaps
                overlapIndex += 1
    overlapIndex

f_resolveFvgDisplay(float zoneTop, float zoneBottom, int overlapIndex) =>
    float displayTop = zoneTop
    float displayBottom = zoneBottom
    int horizontalOffsetBars = 0
    if overlapSpacingMode == OVERLAP_MODE_VERTICAL
        float verticalOffset = overlapIndex * overlapVerticalSpacingTicks * syminfo.mintick
        displayTop := zoneTop - verticalOffset
        displayBottom := zoneBottom - verticalOffset
    else
        horizontalOffsetBars := overlapIndex * overlapHorizontalSpacingBars
    [displayTop, displayBottom, horizontalOffsetBars]

f_drawHtfFvgBlock(string tf, bool isBullish, float zoneTop, float zoneBottom, int horizontalOffsetBars, color textColor, string textSize) =>
    int leftBar = math.min(bar_index + htfFvgRightOffsetBars + horizontalOffsetBars, bar_index + 499)
    int rightBar = math.min(leftBar + htfFvgBlockWidth, bar_index + 500)
    color blockColor = isBullish ? htfFvgBullBgColor : htfFvgBearBgColor
    string blockText = htfFvgTextPrefix + " " + f_formatTfLabel(tf)
    box fvgBox = box.new(left = leftBar, top = zoneTop, right = rightBar, bottom = zoneBottom, xloc = xloc.bar_index, bgcolor = blockColor, border_color = htfFvgBorderColor, border_width = htfFvgBorderWidth, text = blockText, text_color = textColor, text_size = textSize, text_halign = text.align_center, text_valign = text.align_center)
    array.push(htfFvgBoxes, fvgBox)
